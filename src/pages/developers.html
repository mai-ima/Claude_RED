Performance API

ドキュメント
PerformanceSession.kt
// 目標165fpsの性能契約を宣言
val session = SuzakuGameKit.performance(context)
session.requestContract {
    targetFps = 165
    mode = PerfMode.SUZAKU   // 朱雀モード
    allowFan = true
}
session.onThermalHeadroom { headroom ->
    if (headroom < 0.15f) graphics.reduceParticles()
}
